    14→function getShadeHex(productName: string, shadeName: string): string {
    15→  const prod = products.find(
    16→    (p) => p.name.toLowerCase() === productName.toLowerCase()
    17→  );
    18→  if (!prod) return '#C9A96E';
    19→  const shade = prod.shades.find(
    20→    (s) => s.name.toLowerCase() === shadeName.toLowerCase()
    21→  );
    22→  return shade?.hex ?? '#C9A96E';
    23→}
    24→
    25→/* ── animation variants ─────────────────────────────────────── */
    26→
    27→const slideVariants = {
    28→  enter: (direction: number) => ({
    29→    x: direction > 0 ? '100%' : '-100%',
    30→    opacity: 0,
    31→  }),
    32→  center: {
    33→    x: 0,
    34→    opacity: 1,
    35→  },
    36→  exit: (direction: number) => ({
    37→    x: direction > 0 ? '-100%' : '100%',
    38→    opacity: 0,
    39→  }),
    40→};
    41→
    42→const slideTransition = {
    43→  x: { type: 'spring', stiffness: 300, damping: 30 },
    44→  opacity: { duration: 0.2 },
    45→};

    88→/* ── question wrapper ───────────────────────────────────────── */
    89→
    90→function QuestionGroup({
    91→  question,
    92→  options,
    93→  answerKey,
    94→}: {
    95→  question: string;
    96→  options: string[];
    97→  answerKey: string;
    98→}) {
    99→  const { mirrorAnswers, setMirrorAnswer } = useZaiStore();
   100→  const current = mirrorAnswers[answerKey] ?? '';
   101→
   102→  return (
   103→    <div className="space-y-4">
   104→      <p className="text-xs tracking-editorial uppercase text-zai-ivory/40 mb-3">
   105→        {question}
   106→      </p>
   107→      <div className="space-y-3">
   108→        {options.map((opt) => (

   110→            key={opt}

   112→            selected={current === opt}
   113→            onClick={() => setMirrorAnswer(answerKey, opt)}

   116→      </div>
   117→    </div>
   118→  );
   119→}
